public class ShellFixture {
    public static void main(String[] args) throws Exception {
        switch (args[0]) {
            case "input":
            case "input-flood":
                if (args[0].equals("input-flood")) {
                    for (int i = 0; i < 100000; i++) System.out.print("progress");
                    System.out.flush();
                }
                java.security.MessageDigest digest = java.security.MessageDigest.getInstance("SHA-256");
                byte[] buffer = new byte[8192];
                long received = 0;
                int count;
                while ((count = System.in.read(buffer)) != -1) { digest.update(buffer, 0, count); received += count; }
                System.out.println("Received=" + received);
                System.out.println(java.util.HexFormat.of().formatHex(digest.digest()));
                if (args[0].equals("input-flood") && received != 1048576) System.exit(2);
                System.out.println("Success");
                break;
            case "success":
                System.out.print("Success\n");
                break;
            case "failure":
                System.err.print("Failure [DELETE_FAILED_INTERNAL_ERROR]\n");
                System.exit(1);
                break;
            case "flood":
                for (int i = 0; i < 100000; i++) System.out.print("output");
                break;
            case "wait":
                System.out.println("ready");
                System.out.flush();
                Thread.sleep(60000);
                break;
        }
    }
}
